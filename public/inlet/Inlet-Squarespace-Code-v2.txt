<script>
/* Keep real logo assets; show readable brand names only when loading fails. */
(function () {
  'use strict';
  var root = document.getElementById('inlet-page');
  if (!root) return;
  root.querySelectorAll('img[alt]').forEach(function (image) {
    if (!image.alt || image.getAttribute('data-inl-fallback-bound')) return;
    image.setAttribute('data-inl-fallback-bound', 'true');
    var fallback;
    function showName() {
      if (!fallback) {
        fallback = document.createElement('span');
        fallback.className = 'inl-image-fallback';
        fallback.textContent = image.alt;
        image.insertAdjacentElement('afterend', fallback);
      }
      image.classList.add('inl-logo-failed');
      fallback.hidden = false;
    }
    function showLogo() {
      image.classList.remove('inl-logo-failed');
      if (fallback) fallback.hidden = true;
    }
    image.addEventListener('error', showName);
    image.addEventListener('load', showLogo);
    if (image.complete && image.naturalWidth === 0) showName();
  });
})();
</script>
